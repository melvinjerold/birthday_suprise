<script>

let code="";
let index=0;
let musicStarted=false;

const song1=document.getElementById("song1");
const song2=document.getElementById("song2");

/* 🎵 SAFE MUSIC START */
function startMusic(){
if(!musicStarted){
song1.play().catch(()=>{});
musicStarted=true;
}
}

/* ---------------- LOCK ---------------- */
function press(n){

startMusic();

if(code.length<4){
code+=n;
updateDots();

if(code==="0205"){
goSurprise();
}
}
}

function clearCode(){
code="";
updateDots();
}

function updateDots(){
document.querySelectorAll(".dot").forEach((d,i)=>{
d.classList.toggle("filled", i<code.length);
});
}

/* ---------------- SURPRISE ---------------- */
function goSurprise(){
lock.style.display="none";
surprise.style.display="block";

/* RESET GRID */
grid.innerHTML="";

/* LOAD GRID */
loadGrid();

/* RESET BUTTON */
btn.style.left="50%";
btn.style.top="50%";
btn.classList.remove("heart");
btn.innerText="Open Surprise 🎁";

/* MUSIC CONTINUES */
song1.play().catch(()=>{});

/* BUTTON LOGIC */
startButton();
}

/* GRID */
function loadGrid(){
for(let i=1;i<=50;i++){
let img=document.createElement("img");
img.src="img"+i+".jpeg";
grid.appendChild(img);
}
}

/* BUTTON */
const btn=document.getElementById("btn");

function startButton(){

let attempts = 0;

/* your full messages */
const messages=[
"Try again 😄",
"Catch me 😏",
"Not so fast 😎",
"Keep trying 💪",
"Almost 💖",
"One more 😳",
"Almost there 🔥",
"Ok last 💖",
"Now enter 💖",
"Ready 💖"
];

btn.onmouseenter = function(){

if(attempts < messages.length){

/* ✅ show message based on attempt */
btn.innerText = messages[attempts];

let pos=[
"50% 50%","10% 80%","80% 10%","50% 80%","10% 10%",
"80% 50%","30% 30%","70% 70%","20% 60%","60% 20%"
];

btn.style.left = pos[attempts].split(" ")[0];
btn.style.top = pos[attempts].split(" ")[1];

btn.style.transform = "translate(-50%,-50%)";

attempts++;

}else{

btn.classList.add("heart");
btn.innerText = "";
btn.onclick = startFinal;
}
};
}

/* ---------------- FINAL ---------------- */
function startFinal(){

surprise.style.display="none";
final.style.display="block";

/* SWITCH MUSIC */
song1.pause();
song2.play().catch(()=>{});

loadBg();
startBg();
startSlide();
startTyping();
}

/* BACKGROUND */
function loadBg(){
bg.innerHTML="";
for(let i=1;i<=50;i++){
let img=document.createElement("img");
img.src="img"+i+".jpeg";
bg.appendChild(img);
}
}

function startBg(){
let imgs=document.querySelectorAll(".bg img");

index=0;
imgs[0].classList.add("active");

setInterval(()=>{
imgs.forEach(i=>i.classList.remove("active"));
index=(index+1)%imgs.length;
imgs[index].classList.add("active");
},2000);
}

/* SLIDESHOW */
function startSlide(){

let a=document.getElementById("a");
let b=document.getElementById("b");

index=0;

a.src="img1.jpeg";
b.src="img1.jpeg";

a.classList.add("show");

setInterval(()=>{

let src="img"+(index+1)+".jpeg";

if(a.classList.contains("show")){
b.src=src;
b.classList.add("show");
a.classList.remove("show");
}else{
a.src=src;
a.classList.add("show");
b.classList.remove("show");
}

index=(index+1)%50;
},2000);
}

/* TYPEWRITER */
function startTyping(){

const msg=`Happy 25th Birthday 💖
You are my everything ❤️
Our journey started from strangers...`;

let i=0;
letter.innerHTML=`<span id="text"></span><span class="cursor"></span>`;
let t=document.getElementById("text");

function type(){
if(i<msg.length){
t.innerHTML+=msg[i++];
setTimeout(type,30);
}
}
type();
}
